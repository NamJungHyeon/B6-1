# [Q02] GOLD·SILVER 등급 회원을 가입일이 빠른 순으로 조회 (WHERE IN + ORDER BY)

-- [Q02] GOLD·SILVER 등급 회원을 가입일이 빠른 순으로 조회 (WHERE IN + ORDER BY)
SELECT id, name, grade, joined_at
FROM member
WHERE grade IN ('GOLD', 'SILVER')
ORDER BY joined_at ASC;
