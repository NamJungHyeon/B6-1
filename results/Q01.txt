# [Q01] 2만 원 이상인 도서를 가격이 비싼 순으로 조회 (WHERE + ORDER BY)

-- [Q01] 2만 원 이상인 도서를 가격이 비싼 순으로 조회 (WHERE + ORDER BY)
SELECT id, title, author, price
FROM book
WHERE price >= 20000
ORDER BY price DESC;
