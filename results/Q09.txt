# [Q09] 한 번도 대여되지 않은 도서 찾기 (LEFT JOIN + IS NULL)

-- [Q09] 한 번도 대여되지 않은 도서 찾기 (LEFT JOIN + IS NULL)
SELECT b.id, b.title, b.author
FROM book b
LEFT JOIN rental r ON r.book_id = b.id
WHERE r.id IS NULL;
