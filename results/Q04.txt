# [Q04] 제목에 '의'가 들어간 도서 검색 (LIKE 부분 일치 검색)

-- [Q04] 제목에 '의'가 들어간 도서 검색 (LIKE 부분 일치 검색)
SELECT id, title, author, published_year
FROM book
WHERE title LIKE '%의%'
ORDER BY published_year DESC;
