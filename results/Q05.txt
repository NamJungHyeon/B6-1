# [Q05] 최근 30일(2026-08-29 ~ 2026-09-28) 대여 기록 (날짜 범위 조회)

-- [Q05] 최근 30일(2026-08-29 ~ 2026-09-28) 대여 기록 (날짜 범위 조회)
SELECT id, member_id, book_id, rented_at, due_date, status
FROM rental
WHERE rented_at BETWEEN '2026-08-29' AND '2026-09-28'
ORDER BY rented_at;
