# [B1-a] JOIN 방식

-- [B1-a] JOIN 방식
SELECT r.id AS rental_id, m.name AS member_name, b.title AS book_title, r.due_date
FROM rental r
INNER JOIN member m ON m.id = r.member_id
INNER JOIN book   b ON b.id = r.book_id
WHERE r.status = 'OVERDUE'
ORDER BY r.id;
