# [Q03] 가장 최근에 발생한 대여 5건 (ORDER BY + LIMIT)

-- [Q03] 가장 최근에 발생한 대여 5건 (ORDER BY + LIMIT)
SELECT id, member_id, book_id, rented_at, status
FROM rental
ORDER BY rented_at DESC
LIMIT 5;
