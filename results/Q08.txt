# [Q08] 모든 회원의 대여 횟수 — 한 번도 안 빌린 회원도 0으로 표시 (LEFT JOIN)

-- [Q08] 모든 회원의 대여 횟수 — 한 번도 안 빌린 회원도 0으로 표시 (LEFT JOIN)
SELECT m.id, m.name, COUNT(r.id) AS rental_count
FROM member m
LEFT JOIN rental r ON r.member_id = m.id
GROUP BY m.id, m.name
ORDER BY rental_count DESC, m.id;
