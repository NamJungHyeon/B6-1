# [Q15] 전체 평균 가격보다 비싼 도서 (스칼라 서브쿼리)

-- [Q15] 전체 평균 가격보다 비싼 도서 (스칼라 서브쿼리)
SELECT id, title, price,
       (SELECT ROUND(AVG(price), 0) FROM book) AS avg_price
FROM book
WHERE price > (SELECT AVG(price) FROM book)
ORDER BY price DESC;
